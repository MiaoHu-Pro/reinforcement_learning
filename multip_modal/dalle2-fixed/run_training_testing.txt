# Train with both natural-image datasets, pretrained CLIP, and the large U-Net.
# `--dataset all` means Flickr8k + Flickr30k; it excludes FashionMNIST.
# The automatic experiment name will be all-preclip-large-lr2e-4.
LARGE_UNET_JOB_ID=$(sbatch --parsable \
  submit-dalle2-train.sh \
  --dataset all \
  --using-pre-CLIP \
  --large-UNet \
  --decoder-lr 2e-4)

# Run these separately to monitor the submitted training job:
# squeue -j "${LARGE_UNET_JOB_ID}"
# tail -f "result_out/dalle2-train-${LARGE_UNET_JOB_ID}.out"
# sstat -j "${LARGE_UNET_JOB_ID}.batch" \
#   --format=JobID,Elapsed,MaxRSS,AveCPU

# Start inference only if training succeeds. Training and inference must use
# only the saved run name; infer.py reloads its exact configuration.
sbatch --dependency="afterok:${LARGE_UNET_JOB_ID}" \
  submit-dalle2-infer.sh \
  --run-name all-preclip-large-lr2e-4 \
  --prior-candidates 8 \
  --prompt "two children are playing outside" \
  --num-images 4 \
  --output generated_images/two-children-playing-large-unet.png


# Additional examples after the all-dataset large-U-Net model is trained.
sbatch submit-dalle2-infer.sh \
  --run-name all-preclip-large-lr2e-4 \
  --prompt "an astronaut standing on an egg" \
  --num-images 4 \
  --output generated_images/large_UNet_astronaut-standing-egg.png

sbatch submit-dalle2-infer.sh \
  --run-name all-preclip-large-lr2e-4 \
  --prompt "a dog running through green grass" \
  --num-images 4 \
  --output generated_images/large_UNet_dog_running.png

sbatch submit-dalle2-infer.sh \
  --run-name all-preclip-large-lr2e-4 \
  --prompt "a man riding a bicycle" \
  --num-images 4 \
  --output generated_images/large_UNet_man_riding.png

sbatch submit-dalle2-infer.sh \
  --run-name all-preclip-large-lr2e-4 \
  --prompt "two children playing outside" \
  --num-images 4 \
  --output generated_images/large_UNet_children_playing.png


# FashionMNIST-only example. `--dataset fashion_mnist` is equivalent.
FASHION_JOB_ID=$(sbatch --parsable \
  submit-dalle2-train.sh \
  --data fashionMNIST)

sbatch --dependency="afterok:${FASHION_JOB_ID}" \
  submit-dalle2-infer.sh \
  --data fashionMNIST \
  --prompt "An image of a sneaker" \
  --num-images 4 \
  --output generated_images/fashion-mnist-sneaker.png



# Dataset behavior:
#   no dataset option     -> Flickr30k only
#   --dataset all         -> Flickr8k + Flickr30k only
#   --dataset flickr8k    -> Flickr8k only
#   --data fashionMNIST   -> FashionMNIST only
#
# Existing checkpoints are protected. Add --resume to the training command to
# skip complete stages, or --overwrite to explicitly replace them.
